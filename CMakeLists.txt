# IDEs can open the Linux checkout with just a supported CMake and the system prerequisites.
# Explicit compiler and cross-toolchain settings retain their normal CMake semantics.
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux" AND NOT CMAKE_TOOLCHAIN_FILE AND
   (NOT CMAKE_C_COMPILER OR CMAKE_C_COMPILER STREQUAL "clang") AND
   (NOT CMAKE_CXX_COMPILER OR CMAKE_CXX_COMPILER STREQUAL "clang++"))
  cmake_host_system_information(RESULT nw_cpu QUERY OS_PLATFORM)
  if(nw_cpu STREQUAL "x86_64" AND (CMAKE_C_COMPILER OR ("$ENV{CC}" STREQUAL "" AND "$ENV{CXX}" STREQUAL "")))
    find_program(nw_clang NAMES clang NO_CACHE)
    find_program(nw_clangxx NAMES clang++ NO_CACHE)
    if(nw_clang)
      execute_process(COMMAND "${nw_clang}" -dumpversion OUTPUT_VARIABLE nw_clang_version OUTPUT_STRIP_TRAILING_WHITESPACE)
    endif()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/dependencies.lock.json" nw_lock)
    string(JSON nw_llvm_version GET "${nw_lock}" toolchain llvm version)
    if(NOT nw_clangxx OR NOT nw_clang_version VERSION_EQUAL nw_llvm_version)
      find_program(nw_python NAMES python3 REQUIRED)
      message(STATUS "Preparing the locked Korda Linux toolchain in artifacts/toolchain")
      execute_process(COMMAND "${nw_python}" eng/cpp_toolchain.py WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                      OUTPUT_VARIABLE nw_tools COMMAND_ERROR_IS_FATAL ANY)
      string(JSON nw_bin GET "${nw_tools}" bin llvm)
      string(JSON nw_ninja_dir GET "${nw_tools}" bin ninja)
      set(nw_clang "${nw_bin}/clang")
      set(nw_clangxx "${nw_bin}/clang++")
      if(CMAKE_GENERATOR STREQUAL "Ninja" AND NOT CMAKE_MAKE_PROGRAM)
        set(CMAKE_MAKE_PROGRAM "${nw_ninja_dir}/ninja" CACHE FILEPATH "Korda Ninja")
      endif()
    endif()
    set(CMAKE_C_COMPILER "${nw_clang}" CACHE FILEPATH "Korda C compiler" FORCE)
    set(CMAKE_CXX_COMPILER "${nw_clangxx}" CACHE FILEPATH "Korda C++ compiler" FORCE)
  endif()
endif()

if(NW_BUILD_POSTGRESQL)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT PostgreSQL_ROOT)
    cmake_path(CONVERT "$ENV{CMAKE_PREFIX_PATH}" TO_CMAKE_PATH_LIST pg_environment_prefixes)
    foreach(prefix IN LISTS CMAKE_PREFIX_PATH pg_environment_prefixes)
      if(EXISTS "${prefix}/korda-postgresql-build.json")
        set(PostgreSQL_ROOT "${prefix}")
        break()
      endif()
    endforeach()
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT PostgreSQL_ROOT)
    set(pg_root "${CMAKE_SOURCE_DIR}/artifacts/postgres-client")
    execute_process(COMMAND "${CMAKE_COMMAND}" -E env
      "CC=${NW_C_COMPILER_REAL}" "CXX=${NW_CXX_COMPILER_REAL}" "NW_POSTGRES_TOOLS_ROOT=${pg_root}"
      bash "${CMAKE_SOURCE_DIR}/eng/postgres-tools.sh" client COMMAND_ERROR_IS_FATAL ANY)
    set(PostgreSQL_ROOT "${pg_root}/install")
  endif()
  find_package(PostgreSQL 18.0 EXACT REQUIRED)
  find_library(NW_LIBPQ_STATIC NAMES libpq.a HINTS "${PostgreSQL_LIBRARY_DIRS}" REQUIRED NO_DEFAULT_PATH)
  find_library(NW_PGCOMMON_SHARED_STATIC NAMES libpgcommon_shlib.a HINTS "${PostgreSQL_LIBRARY_DIRS}" REQUIRED NO_DEFAULT_PATH)
  find_library(NW_PGPORT_SHARED_STATIC NAMES libpgport_shlib.a HINTS "${PostgreSQL_LIBRARY_DIRS}" REQUIRED NO_DEFAULT_PATH)

    get_filename_component(NW_PG_PREFIX "${PostgreSQL_LIBRARY_DIRS}" DIRECTORY)
    set(NW_PG_PROVENANCE "${NW_PG_PREFIX}/korda-postgresql-build.json")
    if(NOT EXISTS "${NW_PG_PROVENANCE}")
      message(FATAL_ERROR "Build the pinned, patched libpq with eng/postgres-tools.sh and select its prefix")
    endif()
    file(READ "${NW_PG_PROVENANCE}" NW_PG_BUILD_JSON)
    file(SHA256 "${CMAKE_SOURCE_DIR}/native/cpp/patches/libpq-explicit-configuration.patch" NW_EXPECTED_PATCH)
    file(SHA256 "${NW_C_COMPILER_REAL}" NW_EXPECTED_COMPILER)
    string(JSON NW_ACTUAL_PATCH GET "${NW_PG_BUILD_JSON}" patchSha256)
    string(JSON NW_ACTUAL_COMPILER GET "${NW_PG_BUILD_JSON}" compilerSha256)
    string(JSON NW_ACTUAL_ARCHIVE GET "${NW_PG_BUILD_JSON}" archiveSha256)
    if(NOT NW_ACTUAL_PATCH STREQUAL NW_EXPECTED_PATCH OR NOT NW_ACTUAL_COMPILER STREQUAL NW_EXPECTED_COMPILER OR
       NOT NW_ACTUAL_ARCHIVE STREQUAL "0d5b903b1e5fe361bca7aa9507519933773eb34266b1357c4e7780fdee6d6078")
      message(FATAL_ERROR "libpq source, explicit-configuration patch or Clang provenance differs")
    endif()
    foreach(archive IN ITEMS NW_LIBPQ_STATIC NW_PGCOMMON_SHARED_STATIC NW_PGPORT_SHARED_STATIC)
      get_filename_component(archive_name "${${archive}}" NAME)
      file(SHA256 "${${archive}}" archive_sha)
      string(JSON recorded_sha GET "${NW_PG_BUILD_JSON}" files "install/lib/${archive_name}")
      if(NOT archive_sha STREQUAL recorded_sha)
        message(FATAL_ERROR "libpq dependency changed after its source build: ${archive_name}")
      endif()
    endforeach()
  endif()
  find_package(ZLIB REQUIRED)
endif()

function(nw_database_target name sqlite postgres)
  add_library(${name} SHARED $<TARGET_OBJECTS:nw_database_common> ${NW_HOST_SOURCES})
  target_link_libraries(${name} PRIVATE nw_project_options)
  target_include_directories(${name} PRIVATE "${CMAKE_BINARY_DIR}/generated")
  target_compile_definitions(${name} PRIVATE NW_ENABLE_SQLITE=${sqlite} NW_ENABLE_POSTGRES=${postgres})
  if(sqlite)
    target_link_libraries(${name} PRIVATE nw_sqlite_engine)
  endif()
  if(postgres)
    target_include_directories(${name} SYSTEM PRIVATE ${PostgreSQL_INCLUDE_DIRS})
    target_link_libraries(${name} PRIVATE "${NW_LIBPQ_STATIC}" "${NW_PGCOMMON_SHARED_STATIC}" "${NW_PGPORT_SHARED_STATIC}" OpenSSL::SSL ZLIB::ZLIB ${CMAKE_DL_LIBS})
  endif()
  if(UNIX AND NOT APPLE)
    target_link_options(${name} PRIVATE "LINKER:--exclude-libs,ALL" "LINKER:--version-script=${CMAKE_SOURCE_DIR}/native/cpp/exports.map")
    if(NOT NW_SANITIZERS)
      target_link_options(${name} PRIVATE "LINKER:--no-undefined")
    endif()
  endif()
endfunction()
