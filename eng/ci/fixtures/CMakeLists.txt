cmake_minimum_required(VERSION 3.24)
project(InteropFixtureChecks LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(native "${root}/native/cpp")
add_library(interop STATIC "${native}/src/interop.cpp")
target_include_directories(interop PUBLIC "${root}/native/interop/abi/include" "${native}/include")
find_package(Threads REQUIRED)
target_link_libraries(interop PUBLIC Threads::Threads)

add_executable(abi-bootstrap "${root}/tests/Interop.Abi/native/bootstrap.c" "${native}/tests/fixtures/abi.cpp")
target_link_libraries(abi-bootstrap PRIVATE interop)
target_compile_options(abi-bootstrap PRIVATE -UNDEBUG)

option(CHECK_INTEROP_CORPUS "Build the checked-in interop corpus consumer" OFF)
if(CHECK_INTEROP_CORPUS)
  include(FetchContent)
  file(READ "${root}/native/dependencies.lock.json" lock)
  string(JSON count LENGTH "${lock}" dependencies)
  math(EXPR last "${count} - 1")
  foreach(i RANGE 0 ${last})
    string(JSON name GET "${lock}" dependencies ${i} name)
    if(name STREQUAL "nlohmann-json")
      string(JSON url GET "${lock}" dependencies ${i} url)
      string(JSON hash GET "${lock}" dependencies ${i} archiveSha256)
      FetchContent_Declare(json URL "${url}" URL_HASH "SHA256=${hash}")
      FetchContent_MakeAvailable(json)
      break()
    endif()
  endforeach()
  find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
  add_executable(interop-corpus "${root}/tests/Verification.Interop/fuzz/interop.c"
    "${native}/src/contracts.cpp" "${native}/tests/fixtures/verification.cpp" "${native}/tests/fixtures/memory.cpp")
  target_compile_options(interop-corpus PRIVATE -UNDEBUG -fsanitize=address,undefined)
  target_link_options(interop-corpus PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(interop-corpus PRIVATE interop OpenSSL::Crypto nlohmann_json::nlohmann_json)
endif()
