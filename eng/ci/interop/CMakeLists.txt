cmake_minimum_required(VERSION 4.3.3)
project(InteropConformance LANGUAGES C CXX)
get_filename_component(PROJECT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
include(FetchContent)
file(READ "${PROJECT_SOURCE_DIR}/native/dependencies.lock.json" lock)
string(JSON count LENGTH "${lock}" dependencies)
math(EXPR last "${count} - 1")
foreach(i RANGE 0 ${last})
  string(JSON name GET "${lock}" dependencies ${i} name)
  if(name STREQUAL "nlohmann-json")
    string(JSON url GET "${lock}" dependencies ${i} url)
    string(JSON hash GET "${lock}" dependencies ${i} archiveSha256)
    FetchContent_Declare(nw_json URL "${url}" URL_HASH "SHA256=${hash}")
  endif()
endforeach()
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(nw_json)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(nw_project_options INTERFACE)
target_include_directories(nw_project_options INTERFACE "${PROJECT_SOURCE_DIR}/native/cpp/include" "${PROJECT_SOURCE_DIR}/native/interop/abi/include")
target_link_libraries(nw_project_options INTERFACE nlohmann_json::nlohmann_json OpenSSL::Crypto Threads::Threads)
file(WRITE "${CMAKE_BINARY_DIR}/compiler-provenance.txt"
  "C=${CMAKE_C_COMPILER}\nCXX=${CMAKE_CXX_COMPILER}\nID=${CMAKE_CXX_COMPILER_ID}\nVERSION=${CMAKE_CXX_COMPILER_VERSION}\n")
set(NW_INTEROP_ONLY ON)
enable_testing()
add_subdirectory("${PROJECT_SOURCE_DIR}/native/cpp/tests" contract)
